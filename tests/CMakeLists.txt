cmake_minimum_required(VERSION 3.22)
project(hichord_dsp_tests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(dsp_test dsp_test.cpp)
target_compile_features(dsp_test PRIVATE cxx_std_17)
target_include_directories(dsp_test PRIVATE ../app/src/main/cpp)
target_link_libraries(dsp_test PRIVATE Threads::Threads)
option(ENABLE_TSAN "Enable ThreadSanitizer" OFF)
if(ENABLE_TSAN)
    target_compile_options(dsp_test PRIVATE -fsanitize=thread -g)
    target_link_options(dsp_test PRIVATE -fsanitize=thread)
endif()
add_test(NAME dsp_test COMMAND dsp_test)
